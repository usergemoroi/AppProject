```javascript
const functions = require('firebase-functions');

admin.initializeApp();

// НАСТРОЙТЕ ВРЕМЯ ЗАПУСКА БОМБЫ:
exports.scheduleDailyBomb = functions.pubsub
  .schedule('0 19 * * *')  // 19:00 каждый день
  .timeZone('Europe/Moscow')  // Ваш часовой пояс
  .onRun(async (context) => {
    // Логика запуска новой бомбы
    const sessionId = `bomb_${Date.now()}`;
    
    await admin.firestore().collection('bomb_sessions').doc(sessionId).set({
      // ... данные сессии
    });
    
    // Отправка пуша всем
    await admin.messaging().sendToTopic('all_users', {
      notification: {
        title: '💣 БОМБА ЗАПУЩЕНА!',
        body: 'У тебя есть 60 секунд!'
      }
    });
  });

// ИЗМЕНЯЙТЕ:
// - Время: '0 19 * * *' = каждый день в 19:00
// - Формат: 'минуты часы день месяц день_недели'
// - Примеры:
//   '30 20 * * *' = 20:30 каждый день
//   '0 12 * * 1-5' = 12:00 в будни
//   '0 */3 * * *' = каждые 3 часа
```

## 🔐 8. БЕЗОПАСНОСТЬ И ПРАВИЛА

### 8.1 Firestore Rules

**В Firebase Console:** Firestore Database → Rules

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // БОМБЫ: все читают, только владелец создает
    match /bombs/{bombId} {
      allow read: if true;
      allow create: if request.resource.data.deviceId == request.auth.uid
                    || request.resource.data.deviceId != null;
      allow update, delete: if false;
    }
    
    // ГОЛОСА: один голос на бомбу от пользователя
    match /votes/{voteId} {
      allow read: if true;
      allow create: if request.auth != null 
                    && !exists(/databases/$(database)/documents/votes/$(request.auth.uid + '_' + request.resource.data.bombId));
      allow update, delete: if false;
    }
    
    // ПРОФИЛИ: только владелец читает и пишет
    match /users/{userId} {
      allow read, write: if request.auth != null 
                         && request.auth.uid == userId;
    }
  }
}
```

### 8.2 Storage Rules

**В Firebase Console:** Storage → Rules

```javascript
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /bombs/{sessionId}/{fileName} {
      // Разрешить загрузку только изображений и видео до 50MB
      allow read: if true;
      allow write: if request.resource.size < 50 * 1024 * 1024
                   && (request.resource.contentType.matches('image/.*')
                      || request.resource.contentType.matches('video/.*'));
    }
  }
}
```

## 🆘 ЧАСТЫЕ ПРОБЛЕМЫ

### Проблема 1: "google-services.json not found"

**Решение:** Убедитесь что файл в папке `app/` (не в корне проекта)

### Проблема 2: "Firebase initialization failed"

**Решение:** 
1. Проверьте что package name в `google-services.json` совпадает с `applicationId`
2. Синхронизируйте Gradle: File → Sync Project with Gradle Files

### Проблема 3: "Native library not found"

**Решение:**
1. Build → Refresh Linked C++ Projects
2. Build → Clean Project
3. Build → Rebuild Project

### Проблема 4: Приложение падает при старте

**Решение:**
1. Проверьте логи: `adb logcat`
2. Убедитесь что Firebase инициализирован
3. Проверьте что все зависимости скачаны
